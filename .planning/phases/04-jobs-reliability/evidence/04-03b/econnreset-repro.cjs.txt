// Reproduces the server-side error a Node HTTP server sees when the CLIENT closes the socket while the server is still
// reading a request body (a browser context closed with a POST in flight). Loopback only.
const http = require("node:http");
const net = require("node:net");
const server = http.createServer((req, res) => {
  req.on("error", (err) => {
    const frames = String(err.stack).split("\n").slice(1).map((l) => l.trim().replace(/^at /, "")).map((l) => l.replace(/:\d+\)?$/, (m) => m));
    console.log(JSON.stringify({ name: err.name, code: err.code, message: err.message, frames: String(err.stack).split("\n").slice(1).map((l) => l.trim()) }));
    server.close();
  });
  req.on("data", () => {});
});
server.listen(0, "127.0.0.1", () => {
  const { port } = server.address();
  const s = net.connect(port, "127.0.0.1", () => {
    s.write("POST /x HTTP/1.1\r\nHost: 127.0.0.1\r\nContent-Type: text/plain\r\nContent-Length: 1000\r\n\r\npartial");
    setTimeout(() => s.destroy(), 50);
  });
});
