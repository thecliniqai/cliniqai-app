-- 3. change_member_role
create or replace function public.change_member_role(p_membership_id uuid, p_role text)
returns public.memberships language plpgsql security definer set search_path = '' as $$
declare v_uid uuid := auth.uid(); v_target public.memberships; v_caller_role text; v_owners int;
begin
  if v_uid is null then raise exception 'not_authenticated' using errcode = '28000'; end if;
  if p_role not in ('owner','admin','doctor','receptionist','nurse') then
    raise exception 'invalid_role' using errcode = '22023'; end if;
  select * into v_target from public.memberships where id = p_membership_id for update;
  if not found then raise exception 'not_found' using errcode = 'P0002'; end if;
  select role into v_caller_role from public.memberships
   where tenant_id = v_target.tenant_id and user_id = v_uid;
  if v_caller_role is null then raise exception 'not_found' using errcode = 'P0002'; end if; -- no tenant leak
  if v_caller_role not in ('owner','admin') then raise exception 'forbidden' using errcode = '42501'; end if;
  if (p_role in ('owner','admin') or v_target.role in ('owner','admin')) and v_caller_role <> 'owner' then
    raise exception 'owner_required' using errcode = '42501'; end if;
  if v_target.role = 'owner' and p_role <> 'owner' then
    perform 1 from public.memberships where tenant_id = v_target.tenant_id and role = 'owner' for update;
    select count(*) into v_owners from public.memberships where tenant_id = v_target.tenant_id and role = 'owner';
    if v_owners <= 1 then raise exception 'last_owner' using errcode = 'P0001'; end if;
  end if;
  update public.memberships set role = p_role where id = p_membership_id returning * into v_target;
  return v_target;
end; $$;
revoke execute on function public.change_member_role(uuid, text) from public, anon;
grant execute on function public.change_member_role(uuid, text) to authenticated;
