## 1. Executive Summary & Purpose

In Indian outpatient (OPD) healthcare, mobile phone numbers serve as the universal primary patient identifier and the sole digital delivery pipeline for prescriptions, receipts, WhatsApp notifications, and queue token updates. 

Previously, across the application, phone numbers were entered using unformatted, raw `<input type="tel">` text boxes. This allowed inconsistent user entries: omitting the country prefix, including inconsistent prefixes (`+91`, `91`, `0`), entering hyphens/spaces, or entering invalid 8-digit strings.

**Phase 05.2 enforces end-to-end standardization:**
1. **Utility & Domain Foundation:** `lib/utils/phone.ts` enforces the Indian National Numbering Plan (NNP) (`/^[6-9]\d{9}$/`), strips formatting artifacts, and normalizes all persistent records to ITU-T E.164 (`+91XXXXXXXXXX`).
2. **Accessible UI Component:** `components/ui/phone-input.tsx` encapsulates the light-only `DESIGN.md` tokens, an inset Indian flag (`🇮🇳`), a non-editable `+91` badge with divider line, real-time numeric character filtering, length capping at 10 digits, and paste cleanup.
3. **Application-Wide Form Upgrade:** Upgrades all 8 identified phone input locations across patient intake, scheduling drawers, staff management, and settings forms.
4. **Flexible Search & Deduplication:** Upgrades patient and visit search queries so staff can search by bare 10-digit mobile numbers or partial numeric substrings without typing `+91`.
5. **Database Migration:** An idempotent migration (`000017_normalize_indian_phone_numbers.sql`) cleans existing rows across `patients`, `staff_members`, `clinics`, and `communication_queue`.

#### Target Pattern: `lib/utils/phone.ts`
```typescript
export const INDIAN_MOBILE_REGEX = /^[6-9]\d{9}$/;

/**
 * Strips all non-digit characters and extracts the 10-digit Indian subscriber number.
 * Handles inputs with "+91", "91", leading "0", or formatted dashes/spaces.
 */
export function extractTenDigits(phone: string | null | undefined): string {
  if (!phone) return "";
  const digits = phone.replace(/\D/g, "");
  if (digits.length === 10) return digits;
  if (digits.length === 11 && digits.startsWith("0")) return digits.slice(1);
  if (digits.length === 12 && digits.startsWith("91")) return digits.slice(2);
  if (digits.length > 10) return digits.slice(-10);
  return digits;
}

/**
 * Normalizes input to international ITU-T E.164 format (+91XXXXXXXXXX).
 */
export function toE164(phone: string | null | undefined): string {
  const digits = extractTenDigits(phone);
  if (digits.length === 10 && INDIAN_MOBILE_REGEX.test(digits)) {
    return `+91${digits}`;
  }
  return phone?.trim() || "";
}

/**
 * Validates whether the given string is a valid 10-digit Indian mobile number.
 */
export function isValidIndianMobile(phone: string | null | undefined): boolean {
  if (!phone) return false;
  const digits = extractTenDigits(phone);
  return digits.length === 10 && INDIAN_MOBILE_REGEX.test(digits);
}

/**
 * Formats a 10-digit number for display with standard grouping (5+5):
 * "9876543210" -> "98765 43210"
 */
export function formatTenDigitsDisplay(phone: string | null | undefined): string {
  const digits = extractTenDigits(phone);
  if (digits.length <= 5) return digits;
  return `${digits.slice(0, 5)} ${digits.slice(5, 10)}`;
}
```

```tsx
import React, { useState, useEffect } from "react";
import { cn } from "@/lib/utils";
import { extractTenDigits, toE164 } from "@/lib/utils/phone";

export interface PhoneInputProps
  extends Omit<React.InputHTMLAttributes<HTMLInputElement>, "value" | "onChange"> {
  value?: string;
  onChange?: (tenDigits: string, e164: string) => void;
  invalid?: boolean;
}

export const PhoneInput = React.forwardRef<HTMLInputElement, PhoneInputProps>(
  ({ value = "", onChange, invalid = false, disabled = false, className, id, ...props }, ref) => {
    const [digits, setDigits] = useState<string>(() => extractTenDigits(value));

    useEffect(() => {
      setDigits(extractTenDigits(value));
    }, [value]);

    const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
      const raw = e.target.value;
      const cleaned = raw.replace(/\D/g, "").slice(0, 10);
      setDigits(cleaned);
      onChange?.(cleaned, cleaned.length === 10 ? `+91${cleaned}` : cleaned);
    };

    const handlePaste = (e: React.ClipboardEvent<HTMLInputElement>) => {
      e.preventDefault();
      const pasted = e.clipboardData.getData("text");
      const cleaned = extractTenDigits(pasted).slice(0, 10);
      setDigits(cleaned);
      onChange?.(cleaned, cleaned.length === 10 ? `+91${cleaned}` : cleaned);
    };

    return (
      <div
        className={cn(
          "relative flex items-center w-full min-h-[44px] rounded-lg border bg-white shadow-theme-xs transition-colors",
          "focus-within:outline-none focus-within:ring-3 focus-within:ring-offset-0",
          invalid
            ? "border-[#B42318] focus-within:border-[#B42318] focus-within:ring-[#B42318]/15"
            : "border-gray-300 focus-within:border-[#1769E0] focus-within:ring-[#1769E0]/15",
          disabled && "opacity-50 bg-gray-50 cursor-not-allowed",
          className
        )}
      >
        {/* Fixed Inset Indian Flag & +91 Prefix Badge */}
        <div
          aria-hidden="true"
          className="flex items-center gap-1.5 px-3 py-2 select-none border-r border-gray-200 bg-gray-50/80 rounded-l-lg text-gray-700 font-medium text-sm shrink-0"
        >
          <span className="text-base leading-none">🇮🇳</span>
          <span className="font-semibold text-gray-700 tracking-tight text-xs sm:text-sm">+91</span>
        </div>

        {/* 10-Digit Numeric Input */}
        <input
          {...props}
          ref={ref}
          id={id}
          type="tel"
          inputMode="numeric"
          disabled={disabled}
          maxLength={10}
          value={digits}
          onChange={handleChange}
          onPaste={handlePaste}
          placeholder="98765 43210"
          className="flex-1 w-full bg-transparent px-3 py-2 text-sm text-gray-900 placeholder:text-gray-400 focus:outline-none disabled:cursor-not-allowed"
        />
      </div>
    );
  }
);
```

```sql
-- Migration: 000017_normalize_indian_phone_numbers.sql
-- Description: Idempotently cleans and standardizes phone records in patients, staff_members, clinics,
-- and communication_queue to ITU-T E.164 (+91XXXXXXXXXX) format conforming to Phase 05.2.

CREATE OR REPLACE FUNCTION normalize_indian_phone_e164(raw_phone TEXT)
RETURNS TEXT AS $$
DECLARE
    digits TEXT;
BEGIN
    IF raw_phone IS NULL OR trim(raw_phone) = '' THEN
        RETURN raw_phone;
    END IF;

    -- Strip non-digits
    digits := regexp_replace(raw_phone, '\D', '', 'g');

    -- If 12 digits starting with 91
    IF length(digits) = 12 AND digits LIKE '91%' THEN
        digits := substring(digits FROM 3);
    -- If 11 digits starting with 0
    ELSIF length(digits) = 11 AND digits LIKE '0%' THEN
        digits := substring(digits FROM 2);
    END IF;

    -- If exactly 10 digits starting with [6-9], format as E.164 (+91XXXXXXXXXX)
    IF length(digits) = 10 AND digits ~ '^[6-9]' THEN
        RETURN '+91' || digits;
    END IF;

    -- Fallback to trimmed raw phone if non-standard
    RETURN trim(raw_phone);
END;
$$ LANGUAGE plpgsql IMMUTABLE;

-- 2. Clean existing tables
UPDATE public.patients
SET phone = normalize_indian_phone_e164(phone)
WHERE phone IS NOT NULL AND phone NOT LIKE '+91%';

UPDATE public.staff_members
SET phone = normalize_indian_phone_e164(phone)
WHERE phone IS NOT NULL AND phone NOT LIKE '+91%';

UPDATE public.clinics
SET phone = normalize_indian_phone_e164(phone)
WHERE phone IS NOT NULL AND phone NOT LIKE '+91%';

UPDATE public.communication_queue
SET recipient_phone = normalize_indian_phone_e164(recipient_phone)
WHERE recipient_phone IS NOT NULL AND recipient_phone NOT LIKE '+91%';
```

## 4. Architectural & UI Guardrails (Anti-Patterns to Avoid)

Per `AGENTS.md` and `DESIGN.md`:
1. **Never Introduce Dark Mode / Dark Insets:**
